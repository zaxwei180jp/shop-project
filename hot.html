const API_URL = "/api/products"; const productList = document.getElementById("product-list"); async function loadSaleProducts() { try { const res = await fetch(API_URL); const data = await res.json(); console.log("SALE DATA:", data); // 特價商品 const saleProducts = data.filter((p) => p.isSale === true); renderProducts(saleProducts); } catch (err) { console.error(err); productList.innerHTML = `

無法載入特價商品

`; } } function renderProducts(products) { if (!products.length) { productList.innerHTML = `

目前沒有特價商品

`; return; } productList.innerHTML = products.map((p) => `

${p.name}

NT$ ${p.salePrice} NT$ ${p.price}
`).join(""); } loadSaleProducts();
